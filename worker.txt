function withCors(headers = {}) {
  return { ...baseCorsHeaders, ...headers };
}

function tryParseJSON(text) {
  try { return JSON.parse(text); } catch (_) { return null; }
}

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: withCors({ "Content-Type": "application/json" }),
  });
}

async function supabaseInsert(env, table, data) {
  const res = await fetch(`${env.SUPABASE_URL}/rest/v1/${table}`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "apikey": env.SUPABASE_SERVICE_KEY,
      "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
      "Prefer": "return=representation",
    },
    body: JSON.stringify(data),
  });
  return res.ok ? await res.json() : null;
}

async function supabaseUpsert(env, table, data, onConflict = "sig") {
  const res = await fetch(`${env.SUPABASE_URL}/rest/v1/${table}?on_conflict=${onConflict}`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "apikey": env.SUPABASE_SERVICE_KEY,
      "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
      "Prefer": "return=representation,resolution=merge-duplicates",
    },
    body: JSON.stringify(data),
  });
  return res.ok ? await res.json() : null;
}

async function supabaseQuery(env, table, params = "") {
  const res = await fetch(`${env.SUPABASE_URL}/rest/v1/${table}?${params}`, {
    headers: {
      "apikey": env.SUPABASE_SERVICE_KEY,
      "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
    },
  });
  return res.ok ? await res.json() : [];
}

function getTurnoAR_() {
  const h = (new Date().getUTCHours() + 21) % 24; // UTC-3
  return h >= 14 ? "Tarde" : "Mañana";
}

function getDateAR_() {
  const ar = new Date(Date.now() - 3 * 3600 * 1000);
  return ar.toISOString().split("T")[0]; // YYYY-MM-DD
}

// =========================================================
// PARSEO DE PRODUCTOS
// =========================================================
function parsearProductos(productosRaw) {
  if (!productosRaw) return [];
  return productosRaw.split("\n")
    .map(l => l.trim()).filter(Boolean)
    .map(linea => {
      const match = linea.match(/^(\d+(?:\.\d+)?)\s+(.+)$/);
      return match
        ? { descripcion: match[2].trim(), cantidad: parseFloat(match[1]) }
        : { descripcion: linea, cantidad: 1 };
    });
}

        supabaseInsert(env, "notas_cliente", {
          cliente_id: args.cliente_id,

          vendedor,
          tipo:       "nota_ia",
          resumen:    args.nota.slice(0, 120),
          origen:     "ia",
        }),
      ]);
      return { ok: true };
    }

    if (nombre === "crear_tarea") {

        vendedor,
        cliente_id: args.cliente_id || null,
        tarea:      args.tarea,

// IA — CONSTRUIR SYSTEM PROMPT con contexto real de Supabase

function buildSystemPrompt(ctx) {
  let sys = ctx.prompt_base ||
    "Sos Soporte Mercado Limpio, asistente de ventas de la distribuidora. Hablás en español rioplatense, sos directo y breve (máximo 3-4 oraciones). Nunca inventás datos.";

  // Cobertura general del vendedor
  if (ctx.cobertura) {
    const c = ctx.cobertura;
    sys += `\n\n== COBERTURA DE TU CARTERA ==\n`;
    sys += `Total: ${c.total_clientes} clientes | Al día: ${c.al_dia} | Pendientes: ${c.pendientes} | Críticos: ${c.criticos} | Abandonados: ${c.abandonados} | Sin visitar nunca: ${c.nunca_visitados}\n`;
    sys += `Porcentaje de cobertura: ${c.pct_cobertura}%\n`;
  }

  // Clientes críticos (más urgentes)
  if (ctx.criticos?.length) {
    sys += `\n== CLIENTES CRÍTICOS (sin comprar >30 días) ==\n`;
    ctx.criticos.forEach(c => {
      sys += `- [${c.id}] ${c.nombre} ${c.apellido || ""} | ${c.localidad} | ${c.dias_sin_comprar} días sin comprar\n`;
    });
  }

  // Zonas del vendedor
  if (ctx.zonas?.length) {
    sys += `\n== ZONAS ==\n`;
    ctx.zonas.forEach(z => {
      sys += `- ${z.localidad}: ${z.total_clientes} clientes, ${z.criticos} críticos, promedio ${z.promedio_dias} días sin comprar\n`;
    });
  }

  // Cliente en foco (si se abrió el chat desde una ficha)
  if (ctx.cliente) {
    const c = ctx.cliente;
    sys += `\n== CLIENTE EN FOCO ==\n`;
    sys += `ID: ${c.id} | ${c.nombre} ${c.apellido || ""} | ${c.localidad} | ${c.domicilio}\n`;
    sys += `Días sin comprar: ${c.dias_sin_comprar === 999 ? "nunca compró" : c.dias_sin_comprar + " días"}\n`;
  }

  // Últimos pedidos del cliente en foco
  if (ctx.ultimos_pedidos?.length) {
    sys += `\n== ÚLTIMOS PEDIDOS ==\n`;
    ctx.ultimos_pedidos.forEach(p => {
      const fecha = p.fecha_entrega || "sin fecha";
      const prods = (p.productos || "").replace(/\n/g, " | ").slice(0, 120);
      sys += `- ${fecha}: ${prods}\n`;
    });
  }

  // Notas del equipo sobre el cliente
  if (ctx.notas?.length) {
    sys += `\n== NOTAS DEL EQUIPO ==\n`;
    ctx.notas.forEach(n => {
      const fecha = new Date(n.created_at).toLocaleDateString("es-AR");
      sys += `- [${n.created_by} — ${fecha}]: ${n.nota}\n`;
    });
  }

  // Línea de tiempo de interacciones recientes
  if (ctx.timeline?.length) {
    sys += `\n== LÍNEA DE TIEMPO (interacciones recientes) ==\n`;
    ctx.timeline.forEach(t => {
      const fecha = new Date(t.created_at).toLocaleDateString("es-AR");
      sys += `- ${fecha} [${t.tipo}]: ${t.resumen || ""}\n`;
    });
  }

  // Memoria de conversación anterior
  if (ctx.memoria_previa) {
    sys += `\n== MEMORIA DE CONVERSACIÓN ANTERIOR ==\n${ctx.memoria_previa.resumen}\n`;
    if (ctx.memoria_previa.hechos_clave && Object.keys(ctx.memoria_previa.hechos_clave).length) {
      sys += `Hechos clave recordados: ${JSON.stringify(ctx.memoria_previa.hechos_clave)}\n`;
    }
  }

  sys += `\n== PODÉS HACER ==\nRegistrar visitas, crear notas para el equipo, armar recordatorios. Cuando el vendedor te lo pida, ejecutás la acción y confirmás que lo hiciste.`;

  return sys;
}

export default {
  async fetch(request, env, ctx) {
    try {
      const method = request.method.toUpperCase();
      const url    = new URL(request.url);
      const path   = url.pathname;

      // OPTIONS (CORS preflight)
      if (method === "OPTIONS") {
        return new Response(null, {
          status: 204,
          headers: withCors({
            "Access-Control-Allow-Headers":
              request.headers.get("Access-Control-Request-Headers") || "Content-Type",
          }),
        });
      }

      // =========================================================
      // GET
      // =========================================================
      if (method === "GET") {

        // GET /pedidos?estado=&fecha=
        if (path === "/pedidos") {
          const estado = url.searchParams.get("estado");
          const fecha  = url.searchParams.get("fecha");
          let filtros  = "order=created_at.desc&limit=100";
          if (estado) filtros += `&estado=eq.${estado}`;
          if (fecha)  filtros += `&fecha_entrega=eq.${fecha}`;
          const pedidos = await supabaseQuery(env, "pedidos", filtros);
          return jsonResponse(pedidos);
        }

        // GET /pedido/:id
        if (path.startsWith("/pedido/")) {
          const pedidoId = path.split("/")[2];
          const [pedido, items] = await Promise.all([
            supabaseQuery(env, "pedidos",      `id=eq.${pedidoId}`),
            supabaseQuery(env, "items_pedido", `pedido_id=eq.${pedidoId}&order=id.asc`),
          ]);
          return jsonResponse({ pedido: pedido[0], items });
        }

        // GET /auditoria/:pedidoId
        if (path.startsWith("/auditoria/")) {
          const pedidoId = path.split("/")[2];
          const logs = await supabaseQuery(env, "auditoria",
            `pedido_id=eq.${pedidoId}&order=created_at.asc`);
          return jsonResponse(logs);
        }

        // GET /cobertura/:vendedor → alertas de cobertura de cartera
        if (path.startsWith("/cobertura/")) {
          const vendedor = decodeURIComponent(path.split("/")[2] || "");
          const data     = await supabaseRPC(env, "get_alertas_cobertura", { p_vendedor: vendedor });
          return jsonResponse(data || { pendientes: [], resumen: null });
        }

        // GET /timeline/:clienteId → línea de tiempo de un cliente
        if (path.startsWith("/timeline/")) {
          const clienteId = path.split("/")[2];
          const timeline  = await supabaseQuery(env, "timeline_cliente",
            `cliente_id=eq.${clienteId}&order=created_at.desc&limit=30`);
          return jsonResponse(timeline);

        if (path.startsWith("/tareas/")) {
          const vendedor = decodeURIComponent(path.split("/")[2] || "");
          const tareas   = await supabaseQuery(env, "tareas",
            `vendedor=ilike.${encodeURIComponent(vendedor)}&completada=eq.false&order=vence_at.asc.nullslast`);
          return jsonResponse(tareas);
        }

        // Fallback GET → proxy a GAS
        const upstream = await fetch(GAS_URL + url.search, {
          method: "GET",
          headers: { "Accept": "application/json" },
          redirect: "follow",
        });
        const raw    = await upstream.text();
        const parsed = tryParseJSON(raw);
        return jsonResponse(parsed || { error: "No JSON" });
      }

      // =========================================================
      // POST
      // =========================================================
      if (method === "POST") {
        let bodyText = "";
        let bodyJson = null;
        try {
          bodyText = await request.text();
          bodyJson = tryParseJSON(bodyText);
        } catch (e) {}

        // ── RUTA IA LEGACY (app-pedidos: btnAiOptimize) ──

          const aiRes  = await fetch(
            `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`,
            {
              method: "POST",
              headers: { "Content-Type": "application/json" },

                contents: [{ parts: [{ text: bodyJson.prompt }] }],
                systemInstruction: { parts: [{ text: bodyJson.sys }] },
              }),
            }
          );
          const aiData = await aiRes.json();
          const text   = aiData.candidates?.[0]?.content?.parts?.[0]?.text || "No se pudo procesar.";
          return jsonResponse({ text });

                "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
                "Prefer": "return=minimal,resolution=merge-duplicates",
              },
              body: JSON.stringify({
                vendedor,
                token,
                dispositivo:  dispositivo || null,
                activo:       true,
                updated_at:   new Date().toISOString(),
              }),
            });
            return jsonResponse({ exito: true });
          }

          // ── GET COBERTURA (también accesible por POST) ──
          if (accion === "get_cobertura") {
            const { vendedor } = bodyJson;
            const data = await supabaseRPC(env, "get_alertas_cobertura", { p_vendedor: vendedor });
            return jsonResponse({ exito: true, ...(data || { pendientes: [], resumen: null }) });
          }

          // ── COMPLETAR TAREA ──
          if (accion === "completar_tarea") {
            const { tarea_id } = bodyJson;
            await supabasePatch(env, "tareas", `id=eq.${tarea_id}`, { completada: true });
            return jsonResponse({ exito: true });
          }

          // ── PING (chequeo de conectividad desde app-vendedores) ──
          if (accion === "ping") {
            return jsonResponse({ exito: true, ts: Date.now() });
          }

          // ── REGISTRAR MOVIMIENTO (visita individual desde app-vendedores) ──
          if (accion === "registrar_movimiento") {
            const { vendedor, clienteId, tipo, motivo, observacion } = bodyJson;
            const cliente_id = String(clienteId || "");
            if (!cliente_id || !vendedor) return jsonResponse({ exito: false, error: "Faltan campos" });

            // Mapear tipo GAS → tipo Supabase
            let tipoSB = "visita";
            if (tipo === "pedido") tipoSB = "venta";
            else if (String(motivo || "").toLowerCase().includes("no lleg")) tipoSB = "no_llegue";

            const nota = [motivo, observacion].filter(Boolean).join(" · ") || null;

            await Promise.all([
              supabaseInsert(env, "visitas", { cliente_id, vendedor: String(vendedor).toUpperCase(), tipo: tipoSB, nota }),
              supabaseInsert(env, "timeline_cliente", {
                cliente_id, vendedor: String(vendedor).toUpperCase(),
                tipo: tipoSB, resumen: nota || `Visita: ${tipoSB}`, origen: "manual",
              }),
            ]);
            return jsonResponse({ exito: true });
          }

          // ── REGISTRAR BATCH (múltiples visitas — ej: "No llegué" por zona) ──
          if (accion === "registrar_batch") {
            const { vendedor, items } = bodyJson;
            if (!vendedor || !Array.isArray(items)) return jsonResponse({ exito: false, error: "Faltan campos" });

            const ops = items.flatMap(item => {
              const cliente_id = String(item.clienteId || item.cliente_id || "");
              if (!cliente_id) return [];
              let tipoSB = "visita";
              if (item.tipo === "pedido") tipoSB = "venta";
              else if (String(item.motivo || "").toLowerCase().includes("no lleg")) tipoSB = "no_llegue";
              const nota = [item.motivo, item.observacion].filter(Boolean).join(" · ") || null;
              return [
                supabaseInsert(env, "visitas", { cliente_id, vendedor: String(vendedor).toUpperCase(), tipo: tipoSB, nota }),
                supabaseInsert(env, "timeline_cliente", {
                  cliente_id, vendedor: String(vendedor).toUpperCase(),
                  tipo: tipoSB, resumen: nota || `Visita: ${tipoSB}`, origen: "manual",
                }),
              ];
            });
            await Promise.all(ops);
            return jsonResponse({ exito: true });
          }

          // ── CLIENTES SEARCH (buscador live en app-vendedores) ──
          if (accion === "clientes_search") {
            const { vendedor, q, limit = 8 } = bodyJson;
            if (!vendedor || !q) return jsonResponse({ exito: true, items: [] });
            const items = await supabaseRPC(env, "search_clientes", {
              p_vendedor: vendedor, p_query: q, p_limit: Number(limit) || 8,
            });
            return jsonResponse({ exito: true, items: items || [] });
          }

          // ── CLIENTE LOOKUP (ficha completa + último pedido) ──
          if (accion === "cliente_lookup") {
            const { vendedor, clienteId } = bodyJson;
            if (!vendedor || !clienteId) return jsonResponse({ exito: false, error: "Faltan campos" });
            const data = await supabaseRPC(env, "cliente_lookup", {
              p_vendedor: vendedor, p_cliente_id: String(clienteId),
            });
            if (!data || !data.cliente) return jsonResponse({ exito: false, error: "Cliente no encontrado" });
            return jsonResponse({ exito: true, cliente: data.cliente, ultimoPedido: data.ultimoPedido || null, ticket: data.ticket || "" });
          }

          // ── START JORNADA ──
          if (accion === "start_jornada") {
            const { vendedor, zonas } = bodyJson;
            if (!vendedor) return jsonResponse({ exito: false, error: "Falta vendedor" });

            const today   = getDateAR_();
            const turno   = getTurnoAR_();
            const vend    = String(vendedor).toUpperCase();

            // Verificar si ya hay jornada activa para este vendedor/turno/día
            const existing = await supabaseQuery(env, "jornadas",
              `vendedor=ilike.${encodeURIComponent(vend)}&turno=eq.${encodeURIComponent(turno)}&estado=eq.ACTIVA&inicio_fecha=eq.${today}&limit=1`
            );
            if (existing.length > 0) {
              const j = existing[0];
              return jsonResponse({ exito: true, activa: true, jornadaId: j.jornada_id, turno: j.turno, zonas: j.zonas || [] });
            }

            const jornadaId = `J-${vend}-${today}-${turno.toUpperCase()}-${Date.now()}`;
            await supabaseInsert(env, "jornadas", {
              jornada_id:   jornadaId,
              vendedor:     vend,
              turno,
              zonas:        zonas || [],
              estado:       "ACTIVA",
              inicio_fecha: today,
              inicio_at:    new Date().toISOString(),
            });
            return jsonResponse({ exito: true, activa: true, jornadaId, turno, zonas: zonas || [] });
          }

          // ── GET JORNADA ACTIVA ──
          if (accion === "get_jornada_activa") {
            const { vendedor } = bodyJson;
            if (!vendedor) return jsonResponse({ exito: true, activa: false });

            const today  = getDateAR_();
            const result = await supabaseQuery(env, "jornadas",
              `vendedor=ilike.${encodeURIComponent(vendedor)}&estado=eq.ACTIVA&inicio_fecha=eq.${today}&order=inicio_at.desc&limit=1`
            );
            if (!result.length) return jsonResponse({ exito: true, activa: false });
            const j = result[0];
            return jsonResponse({ exito: true, activa: true, jornadaId: j.jornada_id, turno: j.turno, zonas: j.zonas || [] });
          }

          // ── FINALIZAR JORNADA ──
          if (accion === "finalizar_jornada") {
            const { jornadaId, notas } = bodyJson;
            if (!jornadaId) return jsonResponse({ exito: false, error: "Falta jornadaId" });
            await supabasePatch(env, "jornadas", `jornada_id=eq.${encodeURIComponent(jornadaId)}`, {
              estado: "FINALIZADA",
              fin_at: new Date().toISOString(),
              notas:  notas || null,
            });
            return jsonResponse({ exito: true });
          }

          // ── GET TURNO STATE (mañana finalizada? para lógica de tarde) ──
          if (accion === "get_turno_state") {
            const { vendedor } = bodyJson;
            const today  = getDateAR_();
            const turno  = getTurnoAR_();
            const manana = await supabaseQuery(env, "jornadas",
              `vendedor=ilike.${encodeURIComponent(vendedor || "")}&turno=eq.Mañana&inicio_fecha=eq.${today}&limit=1`
            );
            return jsonResponse({
              exito: true,
              turno,
              manana_finalizada: manana.length > 0 && manana[0].estado === "FINALIZADA",
              manana_activa:     manana.length > 0 && manana[0].estado === "ACTIVA",
            });
          }

          // ── ACCIONES DE DEPÓSITO/FACTURACIÓN (existentes — sin cambio) ──
          const { pedidoId, itemId, usuario } = bodyJson;

          if (accion === "marcar_armado" && pedidoId) {
            await supabasePatch(env, "pedidos", `id=eq.${pedidoId}`, { estado: "armado" });
            await supabaseInsert(env, "auditoria", {
              pedido_id: pedidoId, accion: "pedido_armado",
              usuario: usuario || "deposito", detalle: {},
            });
            return jsonResponse({ exito: true });
          }

          if (accion === "marcar_item_armado" && itemId) {
            const now = new Date().toISOString();
            await supabasePatch(env, "items_pedido", `id=eq.${itemId}`, {
              armado: true, armado_por: usuario || "A", armado_at: now,
            });
            return jsonResponse({ exito: true });
          }

          if (accion === "marcar_item_facturado" && itemId) {
            const now = new Date().toISOString();
            await supabasePatch(env, "items_pedido", `id=eq.${itemId}`, {
              facturado: true, facturado_por: usuario || "C", facturado_at: now,
            });
            return jsonResponse({ exito: true });
          }

          if (accion === "marcar_facturado" && pedidoId) {
            await supabasePatch(env, "pedidos", `id=eq.${pedidoId}`, { estado: "facturado" });
            await supabaseInsert(env, "auditoria", {
              pedido_id: pedidoId, accion: "pedido_facturado",
              usuario: usuario || "facturacion", detalle: {},
            });
            return jsonResponse({ exito: true });
          }

          if (accion === "auditoria" && pedidoId) {
            await supabaseInsert(env, "auditoria", {
              pedido_id: pedidoId,
              accion:    bodyJson.tipo || "accion",
              usuario:   usuario || "sistema",
              detalle:   bodyJson.detalle || {},
            });
            return jsonResponse({ exito: true });
          }
        }

        // ── RUTA PEDIDO (app-pedidos — sin cambio) ──
        if (bodyJson && (bodyJson.fechaEntrega || bodyJson.fecha_entrega || bodyJson.numeroCliente || bodyJson.cliente)) {
          const pedidoRow = {
            sig:            bodyJson.sig || String(Date.now()),
            vendedor:       (bodyJson.nombreVendedor || bodyJson.vendedor || "").toUpperCase(),
            cliente_id:     String(bodyJson.numeroCliente || bodyJson.cliente || ""),
            cliente_nombre: (bodyJson.clienteNombre || bodyJson.cliente_nombre || "").toUpperCase() || null,
            productos_raw:  bodyJson.productos || "",
            observaciones:  bodyJson.observaciones || "",
            fecha_entrega:  bodyJson.fechaEntrega || bodyJson.fecha_entrega || null,
            fecha_pedido:   bodyJson.timestamp ? new Date(bodyJson.timestamp).toISOString() : new Date().toISOString(),
            estado:         "pendiente",
          };

          const sbRes  = await fetch(`${env.SUPABASE_URL}/rest/v1/pedidos?on_conflict=sig`, {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
              "apikey": env.SUPABASE_SERVICE_KEY,
              "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
              "Prefer": "return=representation,resolution=ignore-duplicates",
            },
            body: JSON.stringify(pedidoRow),
          });

          const sbBody = await sbRes.text();

          if (!sbRes.ok) {
            return jsonResponse({
              status: "supabase_error", http: sbRes.status,
              detail: sbBody, row_sent: pedidoRow,
            });
          }

          const parsedSb = tryParseJSON(sbBody);
          const pedidoId = parsedSb?.[0]?.id;

          if (!pedidoId) {
            return jsonResponse({ status: "ok", sig: pedidoRow.sig, pedido_id: null, note: "duplicate" });
          }

          // Insertar items
          const items = parsearProductos(bodyJson.productos);
          if (items.length > 0) {
            const itemsRows = items.map(i => ({ pedido_id: pedidoId, descripcion: i.descripcion, cantidad: i.cantidad }));
            await fetch(`${env.SUPABASE_URL}/rest/v1/items_pedido?pedido_id=eq.${pedidoId}`, {
              method: "DELETE",
              headers: { "apikey": env.SUPABASE_SERVICE_KEY, "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}` },
            });
            await supabaseInsert(env, "items_pedido", itemsRows);
          }

          // GAS fire-and-forget
          ctx.waitUntil(
            fetch(GAS_URL, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: bodyText,
              redirect: "follow",
            }).catch(() => {})
          );

          return jsonResponse({ status: "ok", sig: pedidoRow.sig, pedido_id: pedidoId });
        }

        // ── POST /crear-pedido-faltante ──
        if (path === "/crear-pedido-faltante") {
          const { cliente_id, cliente_nombre, fecha_entrega, items } = bodyJson || {};
          if (!cliente_id || !Array.isArray(items) || !items.length) {
            return jsonResponse({ error: "Faltan datos: cliente_id e items requeridos" }, 400);
          }
          const sig = `falt-${cliente_id}-${fecha_entrega || "nd"}-${Date.now()}`;
          const pedidoRes = await fetch(`${env.SUPABASE_URL}/rest/v1/pedidos`, {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
              "apikey": env.SUPABASE_SERVICE_KEY,
              "Authorization": `Bearer ${env.SUPABASE_SERVICE_KEY}`,
              "Prefer": "return=representation",
            },
            body: JSON.stringify({
              cliente_id,
              cliente_nombre: (cliente_nombre || "").toUpperCase(),
              vendedor: "MERCADO LIMPIO",
              fecha_entrega: fecha_entrega || null,
              observaciones: "Artículos que faltaron en pedidos anteriores",
              estado: "pendiente",
              es_pedido_faltante: true,
              sig,
            }),
          });
          if (!pedidoRes.ok) {
            const err = await pedidoRes.text();
            return jsonResponse({ error: "Error creando pedido: " + err }, 500);
          }
          const pedidoData = await pedidoRes.json();
          const pedidoId = pedidoData?.[0]?.id;
          if (!pedidoId) return jsonResponse({ error: "No se obtuvo id del pedido" }, 500);

          const itemsRows = items.map(it => ({
            pedido_id:  pedidoId,
            descripcion: String(it.descripcion || "").toUpperCase(),
            cantidad:    Number(it.cantidad) || 1,
          }));
          await supabaseInsert(env, "items_pedido", itemsRows);

          return jsonResponse({ ok: true, pedido_id: pedidoId });
        }

        // Fallback → proxy a GAS
        const upstream = await fetch(GAS_URL, {
          method: "POST",
          headers: { "Content-Type": "application/json", "Accept": "application/json" },
          body: bodyText,
          redirect: "follow",
        });
        const raw    = await upstream.text();
        const parsed = tryParseJSON(raw);
        if (!parsed) return jsonResponse({ status: "error", message: "GAS no devolvió JSON." }, 502);
        return jsonResponse(parsed);
      }

      return new Response("Method not allowed", { status: 405, headers: baseCorsHeaders });

    } catch (e) {
      return new Response(JSON.stringify({ error: e.message }), {
        status: 500,
        headers: withCors({ "Content-Type": "application/json" }),
      });
    }
  },
};
