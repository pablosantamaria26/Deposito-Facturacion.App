⚙️ Tabla ai_memoria

La tabla ai_memoria no existe en tu Supabase. Copiá este SQL y ejecutalo en el SQL Editor de Supabase:

-- Ejecutar en Supabase SQL Editor
CREATE TABLE ai_memoria (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  tipo TEXT NOT NULL DEFAULT 'nota',
  clave TEXT,
  contenido TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE ai_memoria ENABLE ROW LEVEL SECURITY;

CREATE POLICY "anon_read"   ON ai_memoria FOR SELECT USING (true);
CREATE POLICY "anon_insert" ON ai_memoria FOR INSERT WITH CHECK (true);
CREATE POLICY "anon_update" ON ai_memoria FOR UPDATE USING (true);
CREATE POLICY "anon_delete" ON ai_memoria FOR DELETE USING (true);
